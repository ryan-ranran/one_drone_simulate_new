#如何运行这个相机（只是在我的笔记本上）
```bash
cd ~/one_drone
source /opt/ros/jazzy/setup.bash
source install/setup.bash
export RMW_IMPLEMENTATION=rmw_fastrtps_cpp

ros2 launch camera_stream stereo.launch.py
```
#如何查看画面
```bash
ros2 run rqt_image_view rqt_image_view /camera/ov9782/left/image_raw
ros2 run rqt_image_view rqt_image_view /camera/ov9782/right/image_raw
```
